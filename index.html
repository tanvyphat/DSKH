Cấu Hình Kết Nối Supabase Database

Nhập Supabase Project URL và Anon Public Key của bạn để lưu trữ trực tiếp trên đám mây. Khi chưa điền, web tự động lưu trong trình duyệt để bạn dùng thử ngay mọi tính năng Thêm, Sửa, Xoá, Lưu, Cập nhật.

Lệnh SQL tạo bảng trong Supabase (SQL Editor) 
create table if not exists public.customers (
  id uuid primary key default gen_random_uuid(),
  category text not null,
  sub_category text,
  store_name text,
  address text,
  phone text,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

alter table public.customers enable row level security;

create policy "Cho phep tat ca thao tac CRUD"
  on public.customers for all
  using (true)
  with check (true);
